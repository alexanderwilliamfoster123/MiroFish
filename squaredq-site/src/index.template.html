Research it, prove it, ship it — without changing languages

The same strategy object drives the simulator and the live trader. Three commands take it from an idea to a paper account.

mean_reversion.py
from squaredq import Strategy, signals

class MeanReversion(Strategy):
    universe = "us_equity_liquid_500"
    schedule = "every 5m during rth"

    def on_bar(self, book):
        z = signals.zscore(book.close, window=60)
        edge = -z.clip(-2, 2)

        self.target_weights(
            edge / edge.abs().sum(),
            max_participation=0.05,
        )
